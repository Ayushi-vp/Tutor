    __slots__ = ('k', 'v', 'prev', 'next')

class LRUCache:
    def __init__(self, cap):
        self.cap, self.map = cap, {}
        self.head, self.tail = Node(), Node()   # sentinels
        self.head.next, self.tail.prev = self.tail, self.head

    def _remove(self, n):
        n.prev.next, n.next.prev = n.next, n.prev

    def _push_front(self, n):
        n.next, n.prev = self.head.next, self.head
        self.head.next.prev = n
        self.head.next = n

    def get(self, k):
        n = self.map.get(k)
        if n is None: return -1
        self._remove(n); self._push_front(n)
        return n.v

    def put(self, k, v):
        if k in self.map:
            n = self.map[k]; n.v = v
            self._remove(n); self._push_front(n); return
        if len(self.map) == self.cap:
            lru = self.tail.prev
            self._remove(lru); del self.map[lru.k]
        n = Node(); n.k, n.v = k, v
        self.map[k] = n; self._push_front(n)

            if not buf: out += split(p, max_tok, enc, depth + 1)
    if buf: out.append(buf)
    return out

What makes it production-grade

  • Overlap by re-prepending the last k tokens of the previous chunk — 10–20% is the usual band; more costs storage and retrieval precision.
  • Atomic blocks: never split a fenced code block or a table row-wise. Extract them first, chunk them whole, and place them back.
  • Heading path prepended to each chunk ("Billing > Refunds > Partial refunds") so an isolated chunk still says what it is about. This alone measurably improves retrieval.
  • Stable chunk ids from a content hash, so re-ingesting an unchanged document does not churn the index.
  • Encoder caching: enc.encode is called constantly in the loop above — memoise per part or the chunker becomes the bottleneck.

Requirements

Accept concurrent single requests; group them into batches of at most B; never make a request wait longer than T ms; return each caller their own result; handle a failing batch without failing unrelated requests.

Design

A single-consumer queue with a dual trigger: flush when the batch is full or when the oldest item hits its deadline. Each request carries a future the caller awaits.

class BatchQueue:
    def __init__(self, fn, max_batch=32, max_wait=0.02):
        self.fn, self.max_batch, self.max_wait = fn, max_batch, max_wait
        self.q = asyncio.Queue()
        asyncio.create_task(self._loop())

    async def submit(self, item):
        fut = asyncio.get_running_loop().create_future()
        await self.q.put((item, fut))
        return await fut

    async def _loop(self):
        while True:
            item, fut = await self.q.get()          # block for the first
            batch, futs = [item], [fut]
            deadline = time.monotonic() + self.max_wait
            while len(batch) < self.max_batch:
                left = deadline - time.monotonic()
                if left <= 0: break
                try:
                    i, f = await asyncio.wait_for(self.q.get(), left)
                except asyncio.TimeoutError:
                    break
                batch.append(i); futs.append(f)
            try:
                for f, r in zip(futs, await self.fn(batch)):
                    if not f.done(): f.set_result(r)
            except Exception as e:
                for f in futs:
                    if not f.done(): f.set_exception(e)

Follow-ups

  • Why block on the first item? An empty-queue spin wastes CPU, and the latency clock should start when work exists, not before.
  • Cancellation: if the caller goes away, mark the future cancelled and skip it at flush — otherwise you pay for abandoned work.
  • Bucketing: for real inference, group by similar sequence length so padding does not dominate the batch cost.
  • Backpressure: a bounded queue that rejects with 429 beats an unbounded one that turns into a latency bomb.
  • Tuning: max_wait is a direct latency-versus-throughput dial; derive it from the p99 target, do not guess.
